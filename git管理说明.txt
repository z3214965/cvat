docker images --format "{{.Repository}}:{{.Tag}}" | grep -v '<none>' | while read image; do
  # 将 / 和 : 替换为 _ ，生成合法文件名
  filename=$(echo "$image" | sed 's/[\/:]/_/g')
  docker save -o "${filename}.tar" "$image"
  echo "✓ 已导出: $image → ${filename}.tar"
done
